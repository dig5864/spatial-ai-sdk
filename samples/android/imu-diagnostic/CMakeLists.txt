cmake_minimum_required(VERSION 3.22.1)
project(spatial_imu_diagnostic LANGUAGES CXX)
if(NOT ANDROID OR NOT EXISTS "${CARDBOARD_ROOT}/sdk/sensors/sensor_fusion_ekf.cc")
  message(FATAL_ERROR "Android and pinned CARDBOARD_ROOT are required")
endif()
file(TO_CMAKE_PATH "${CARDBOARD_ROOT}/sdk" SDK)
file(GLOB FUSION_SOURCES "${SDK}/sensors/*.cc")
file(GLOB MATH_SOURCES "${SDK}/util/*.cc")
add_library(spatial_imu_diagnostic SHARED fusion_jni.cpp ${FUSION_SOURCES} ${MATH_SOURCES})
target_include_directories(spatial_imu_diagnostic PRIVATE "${SDK}")
target_compile_features(spatial_imu_diagnostic PRIVATE cxx_std_17)
target_link_libraries(spatial_imu_diagnostic PRIVATE log)
