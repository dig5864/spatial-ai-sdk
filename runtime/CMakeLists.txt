cmake_minimum_required(VERSION 3.20)
project(spatial_runtime VERSION 0.1.0 LANGUAGES C CXX)
option(SPATIAL_BUILD_TESTS "Build host ABI tests" ON)
add_library(spatial_runtime SHARED core/spatial_runtime.cpp)
target_include_directories(spatial_runtime PUBLIC include)
target_compile_features(spatial_runtime PRIVATE cxx_std_11)
target_compile_definitions(spatial_runtime PRIVATE SPATIAL_RUNTIME_EXPORTS)
set_target_properties(spatial_runtime PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(WIN32)
  set_target_properties(spatial_runtime PROPERTIES PREFIX "")
endif()
if(MINGW)
  target_link_options(spatial_runtime PRIVATE -static-libgcc -static-libstdc++)
endif()
if(SPATIAL_BUILD_TESTS AND NOT ANDROID)
  enable_testing()
  add_executable(spatial_abi_test ../tests/NativeAbi/abi_test.c)
  target_link_libraries(spatial_abi_test PRIVATE spatial_runtime)
  add_test(NAME spatial_abi_contract COMMAND spatial_abi_test)
endif()
